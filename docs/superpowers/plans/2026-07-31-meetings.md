# Meetings for Presentation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the presentation executive (Shayank) record meetings (date + time + remark) against an inquiry from the Inquiry Detail's Presentation section, remove the hardcoded "Today's Meetings" testing data, and show real meetings on the Presentation Dashboard.

**Architecture:** Mirrors the existing Follow-up feature. New `meetings` table keyed by `inquiry_id`; CRUD lives in the existing `/api/inquiries` router; the presentation KPI endpoint gains a real meeting count (`client_meetings_today`) and a `meetings` list (today + next 30 days). Frontend adds an Add Meeting form + list in the Presentation section and the dashboard card shows both Today and Upcoming sections from real data.

**Tech Stack:** FastAPI + SQLAlchemy async + Alembic (backend, PostgreSQL), React + TanStack Query + Tailwind (frontend), Docker Compose for deployment.

**Deployment rules (critical — read before starting):**
- Backend container `shaguncrm-backend-1` has NO persistent code volume. After editing any backend file on the host, copy it in and restart: `docker cp <hostfile> shaguncrm-backend-1:/app/<containerpath>` then `docker restart shaguncrm-backend-1`.
- `docker compose build backend` is broken (EOF error) — do NOT use it. Use `docker cp` + restart.
- Migrations: copy the new migration file into the container, then `docker exec -w /app shaguncrm-backend-1 alembic upgrade head`.
- Frontend: `npm run build` in `D:\Shagun CRM\frontend`, then `docker cp "D:\Shagun CRM\frontend\dist\." shaguncrm-frontend-1:/usr/share/nginx/html/`. Browser hard-refresh with `Ctrl+Shift+R`.
- **Git caution:** The repo has many unrelated modified/untracked files (settlement/dashboard services, alembic env, Templates/, stray migrations). Stage ONLY the files listed in each task's commit step.
- Admin test credentials: `admin@shaguncatering.com` / `admin123`. Presentation exec: `shayank@shayank123`. In PowerShell use `curl.exe` (not `curl`, which is aliased to Invoke-WebRequest).

@router.get("/{inquiry_id}/meetings", response_model=list[MeetingResponse])
async def list_meetings(
    inquiry_id: uuid.UUID,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    await get_inquiry_or_404(db, inquiry_id)
    result = await db.execute(
        select(Meeting)
        .where(Meeting.inquiry_id == inquiry_id)
        .order_by(Meeting.meeting_at.asc())
    )
    return [MeetingResponse.model_validate(m) for m in result.scalars().all()]


@router.post("/{inquiry_id}/meetings", response_model=MeetingResponse, status_code=201)
async def add_meeting(
    inquiry_id: uuid.UUID, data: MeetingCreate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    inquiry = await get_inquiry_or_404(db, inquiry_id)
    meeting = Meeting(
        id=uuid.uuid4(), inquiry_id=inquiry.id,
        meeting_at=data.meeting_at,
        remarks=data.remarks,
        created_by=current_user.id,
    )
    db.add(meeting)
    await db.commit()
    await db.refresh(meeting)
    return MeetingResponse.model_validate(meeting)


@router.patch("/{inquiry_id}/meetings/{meeting_id}", response_model=MeetingResponse)
async def update_meeting_status(
    inquiry_id: uuid.UUID, meeting_id: uuid.UUID, data: MeetingStatusUpdate,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    result = await db.execute(
        select(Meeting).where(Meeting.id == meeting_id, Meeting.inquiry_id == inquiry_id)
    )
    meeting = result.scalar_one_or_none()
    if not meeting:
        raise HTTPException(status_code=404, detail="Meeting not found")
    meeting.status = data.status
    await db.commit()
    await db.refresh(meeting)
    return MeetingResponse.model_validate(meeting)
```

async def get_presentation_kpis(db: AsyncSession, user_id) -> dict:
    today = date.today()
    assigned = (await db.execute(select(func.count(Inquiry.id)).where(
        Inquiry.assigned_to == user_id,
        Inquiry.status.notin_([InquiryStatus.ADVANCE_RECEIVE, InquiryStatus.OPERATION_HANDOVER, InquiryStatus.CANCELLED])
    ))).scalar() or 0
    new_inquiry_count = (await db.execute(select(func.count(Inquiry.id)).where(
        Inquiry.assigned_to == user_id,
        Inquiry.status == InquiryStatus.NEW_INQUIRY
    ))).scalar() or 0
    pending = (await db.execute(select(func.count(Inquiry.id)).where(
        Inquiry.assigned_to == user_id,
        Inquiry.status.in_([InquiryStatus.NEW_INQUIRY, InquiryStatus.FOLLOWUP, InquiryStatus.MENU_SENT])
    ))).scalar() or 0
    meetings_today = (await db.execute(select(func.count(Meeting.id)).where(
        func.date(Meeting.meeting_at) == today
    ))).scalar() or 0
    upcoming_end = today + timedelta(days=30)
    meetings_result = await db.execute(
        select(Meeting, Inquiry.client_name, Inquiry.event_type)
        .join(Inquiry, Meeting.inquiry_id == Inquiry.id)
        .where(func.date(Meeting.meeting_at) >= today, func.date(Meeting.meeting_at) <= upcoming_end)
        .order_by(Meeting.meeting_at.asc())
    )
    meetings = [
        {
            "id": str(m.id),
            "client_name": name,
            "event_type": event_type,
            "meeting_at": m.meeting_at.isoformat(),
            "remarks": m.remarks,
            "status": m.status,
        }
        for m, name, event_type in meetings_result.all()
    ]
    return {
        "new_inquiry": new_inquiry_count,
        "assigned_inquiries": assigned,
        "pending_presentations": pending,
        "client_meetings_today": meetings_today,
        "meetings": meetings,
    }
```
